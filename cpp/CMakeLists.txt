cmake_minimum_required(VERSION 3.16)
project(aetherrun CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(Threads REQUIRED)
file(GLOB CORE_SRC src/*.cpp)
list(REMOVE_ITEM CORE_SRC ${CMAKE_SOURCE_DIR}/src/main.cpp)
add_library(aether_core STATIC ${CORE_SRC})
target_include_directories(aether_core PUBLIC include)
target_link_libraries(aether_core PUBLIC Threads::Threads)
add_executable(aetherrun src/main.cpp)
target_link_libraries(aetherrun aether_core)
enable_testing()
file(GLOB TEST_SRC tests/test_*.cpp)
add_executable(aether_tests tests/main.cpp ${TEST_SRC})
target_compile_definitions(aether_tests PRIVATE AETHER_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data")
target_link_libraries(aether_tests aether_core)
add_test(NAME all COMMAND aether_tests)
